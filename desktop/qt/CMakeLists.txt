cmake_minimum_required(VERSION 3.21)
project(ChipDraw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Qml Svg SvgWidgets)
include(CTest)
qt_add_library(chipdraw_ui STATIC canvas.cpp window.cpp)
target_link_libraries(chipdraw_ui PUBLIC Qt6::Widgets Qt6::Qml Qt6::SvgWidgets)
qt_add_resources(chipdraw_ui model PREFIX "/" FILES chipdraw-model.generated.js)
qt_add_executable(ChipDraw main.cpp)
set_target_properties(ChipDraw PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
target_link_libraries(ChipDraw PRIVATE chipdraw_ui)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(native_test native_test.cpp)
    target_link_libraries(native_test PRIVATE chipdraw_ui Qt6::Test)
    add_test(NAME native_rectangle COMMAND native_test -platform offscreen)
endif()
